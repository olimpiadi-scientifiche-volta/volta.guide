## Binary Exponentiation
```cpp
int exp(int b, int e) {
	if (e == 0) {
		return 1;
	}
	
	int x = exp(b, e / 2);

	if (e % 2 == 0) {
		return x * x;
	}
	if (e % 2 == 1) {
		return x * x * b;
	}
}
```

```cpp
#include <bits/stdc++.h>

using namespace std;

int main() {
	int N; 
	cin >> N;

	vector<int> S(N); 
	for (int i = 0; i < N; i++) {
		cin >> S[i];
	}

	int pari1 = -1, pari2 = -1;
	int dispari1 = -1, dispari2 = -1;

	for (int i = 0; i < N; i++) {
		if (S[i] % 2 == 0) {
			if (S[i] > pari1) {
				pari2 = pari1;
				pari1 = S[i];
			}
			else if (S[i] > pari2) {
				pari2 = S[i];
			}
		}
		if (S[i] % 2 == 1) {
			if (S[i] > dispari1) {
				dispari2 = dispari1;
				dispari1 = S[i];
			}
			else if (S[i] > dispari2) {
				dispari2 = S[i];
			}
		}
	}

	if (pari2 == -1 && dispari2 == -1) {
		cout << -1 << '\n';
	}
	else if (pari2 == -1) {
		cout << dispari1 + dispari2 << '\n';
	}
	else if (dispari2 == -1) {
		cout << pari1 + pari2 << '\n';
	}
	else {
		cout << max(pari1 + pari2, dispari1 + dispari2) << '\n';
	}
}
```
